} /* ========================================================================= PARSEUR MIME ROBUSTE (travaille sur les octets bruts, gère le charset, le multipart imbriqué, quoted-printable et base64, HTML seul). ========================================================================= */ // Lit le flux brut en Uint8Array (octets, sans décodage prématuré) async function streamToBytes(stream) { const reader = stream.getReader(); for (let i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]); return s; } // Convertit une chaîne "binaire" (1 char = 1 octet) en Uint8Array function binStringToBytes(str) { const out = new Uint8Array(str.length); } } return new Uint8Array(bytes); } // Décode base64 -> octets bruts (Uint8Array) function b64ToBytes(str) { try { const bin = atob(str.replace(/\s/g, '')); return binStringToBytes(bin); } catch (e) { return new Uint8Array(0); } } // Parse les en-têtes d'un bloc MIME (déplie les lignes repliées) function parseHeaders(headerText) { const headers = {}; headerText.replace(/\r?\n[ \t]+/g, ' ').split(/\r?\n/).forEach(line => { const m = line.match(/^([^:]+):\s*(.*)$/);
