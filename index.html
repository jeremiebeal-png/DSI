-- ============================================================ -- DSI HOTEL — Suppression définitive d'un ticket (super-admin) -- À exécuter dans le SQL Editor de Supabase. -- -- Supprime, de façon atomique et irréversible : -- - les pièces jointes (ticket_fichiers) -- - les messages (ticket_messages) -- - le ticket lui-même (tickets) -- -- Réservé au super-admin (support@dsihotel.com). La suppression des -- FICHIERS dans le Storage est faite côté portail AVANT l'appel à cette -- fonction (cette fonction ne touche qu'aux lignes de la base). -- ============================================================ -- Vérifie que l'appelant est le super-admin. -- NOTE : l'email n'est PAS stocké dans public.profiles mais dans auth.users -- (c'est ce que le portail utilise : session.user.email). On lit donc l'email -- de l'utilisateur connecté directement depuis auth.users via auth.uid(). create or replace function public.est_super_admin_courant() returns boolean language sql stable security definer set search_path = public as $$ select exists ( select 1 from auth.users u where u.id = auth.uid() and lower(u.email) = 'support@dsihotel.com' ); $$; -- Suppression définitive d'un ticket et de tout ce qui lui est rattaché. -- Renvoie le nombre de messages + fichiers supprimés (pour information). create or replace function public.supprimer_ticket_definitif(p_ticket_id uuid) returns jsonb language plpgsql volatile security definer set search_path = public as $$ declare v_msgs int := 0; v_files int := 0; v_ref text; begin if not public.est_super_admin_courant() then raise exception 'Accès refusé : action réservée au super-administrateur.'; end if; -- Vérifie que le ticket EXISTE (par son id), indépendamment de sa référence. -- Certains anciens tickets de test ont reference = NULL : il ne faut pas les -- considérer comme introuvables. if not exists (select 1 from public.tickets where id = p_ticket_id) then raise exception 'Ticket introuvable.'; end if; select reference into v_ref from public.tickets where id = p_ticket_id; -- Pièces jointes (lignes DB) delete from public.ticket_fichiers where ticket_id = p_ticket_id; get diagnostics v_files = row_count; -- Messages delete from public.ticket_messages where ticket_id = p_ticket_id; get diagnostics v_msgs = row_count; -- Le ticket lui-même delete from public.tickets where id = p_ticket_id; return jsonb_build_object( 'reference', coalesce(v_ref, '(sans référence)'), 'messages_supprimes', v_msgs, 'fichiers_supprimes', v_files ); end; $$; grant execute on function public.est_super_admin_courant() to authenticated; grant execute on function public.supprimer_ticket_definitif(uuid) to authenticated; notify pgrst, 'reload schema';
