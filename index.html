Compléter les lignes déjà présentes (même IP ou même nom) — seulement les champs vides update inventaire_serveurs s set role = coalesce(nullif(s.role, ''), a.role), ip = coalesce(nullif(s.ip, ''), a.ip), commentaire = coalesce(nullif(s.commentaire, ''), a.commentaire), maj_le = now(), maj_par = 'Import DT Boulogne' from _bo a, hotels h where h.name ilike '%boulogne%' and s.hotel_id = h.id and (s.ip = a.ip or lower(s.nom) = lower(a.nom)); -- 2.
